# 10,000 demo:counter blocks (barrel stand-ins ticking once a second through the content host).
# scripts\scenario.ps1 -Scenario tests\bench\m1-counters.txt -RunDir run\bench-counters -Fresh
mods engine\pumpkin\target\release\counter.dll
start
# A player at the centre keeps the 9x9 chunks around it loaded and ticking on every build
# (stock /forceload never loads chunks).
botstart bench
wait 3
cmd tp bench 1074.5 120 1074.5
wait 30
cmd moddedmc place demo:counter 10000 1024 100 1024
# Placing 10,000 in one tick takes ~11 s (bench tool only, not a gameplay path).
wait 20
expect placed 10000/10000 demo:counter
cmd tick query
wait 5
cmd tick query
wait 5
cmd tick query
measure 20
cmd moddedmc dump demo:counter 2
wait 1
expect moddedmc dump: 10000 live demo:counter
bot quit
stop
