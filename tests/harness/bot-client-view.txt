# Observe the initial chunk, block updates, and the lifecycle of a tracked item display.
seed 12345
area 0 150 0 15 153 15
start
cmd fill 0 150 0 15 150 15 minecraft:stone
cmd setblock 2 151 2 minecraft:oak_log[axis=x]
bot join alice
cmd tp alice 1.5 151 1.5 0 0
bot alice await-teleport
tick 2
bot alice expect block 2 151 2 minecraft:oak_log[axis=x]
cmd setblock 2 151 2 minecraft:oak_log[axis=z]
tick 2
bot alice expect block 2 151 2 minecraft:oak_log[axis=z]
cmd summon minecraft:item_display 3.5 151 3.5
tick 2
bot alice expect entity minecraft:item_display near 3.5 151 3.5 within 0.001 count 1
cmd teleport @e[type=minecraft:item_display,limit=1] 4.5 151 3.5
tick 3
bot alice expect entity minecraft:item_display near 4.5 151 3.5 within 0.001
bot alice expect entity minecraft:item_display near 3.5 151 3.5 within 0.001 count 0
cmd kill @e[type=minecraft:item_display]
tick 2
bot alice expect entity minecraft:item_display near 4.5 151 3.5 within 0.001 count 0
stop
