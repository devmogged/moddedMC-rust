# Verify that vehicle movement packets move the server's mount across the test floor.
seed 12345
area 0 150 0 15 155 10
start
cmd fill 0 150 0 15 150 10 minecraft:stone
bot join alice
cmd tp alice 1.5 151 1.5 0 0
bot alice await-teleport
cmd summon minecraft:oak_boat 1.5 151 2.8
bot alice interact minecraft:oak_boat 1.5 151 2.8 main
bot alice expect riding minecraft:oak_boat
bot alice physics manual
bot alice input forward
bot alice physics-tick 3
probe execute if entity @e[type=minecraft:oak_boat,x=1.5,y=151,z=2.9824,distance=..0.00001]
expect Test passed
