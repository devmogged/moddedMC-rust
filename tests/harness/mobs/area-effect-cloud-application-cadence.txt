# Java 26.3 AreaEffectCloud.serverTick applies potion effects only every five ticks.
seed 12345
area 0 150 0 7 155 7
gamerules deterministic
start
cmd fill 0 150 0 7 150 7 minecraft:stone
cmd summon minecraft:creeper 2.5 151 2.5 {NoAI:1b,NoGravity:1b,ignited:1b,Fuse:1s,active_effects:[{id:"minecraft:speed",duration:200,amplifier:0}]}
tick 4
tick 8
cmd summon minecraft:pig 2.5 151 2.5 {NoAI:1b,NoGravity:1b,Invulnerable:1b}
tick 1
probe data get entity @e[type=minecraft:pig,limit=1] active_effects
stop
