# Mod data: a .mmc's data/ loads as a built-in datapack (recipes, loot tables).
# Build with scripts/pack-mod.ps1 -Crate create and scripts/engine.ps1 build.
# Run scripts/scenario.ps1 -Scenario tests/scenarios/create-data.txt -RunDir run/create-data -Fresh
mods target\mods\create.mmc
start
expect Loaded built-in datapack 'mod/create': 3 recipe\(s\), 4 loot table\(s\)
botstart tester
wait 3
expect \[bot\] joined as tester
cmd tp tester 1025.5 101 1023.5
wait 3
cmd recipe give tester create:crafting/kinetics/cogwheel
wait 1
expect Unlocked 1 recipe
cmd setblock 1025 100 1025 create:shaft
wait 1
cmd loot spawn 1025 101 1025 mine 1025 100 1025
wait 1
expect Dropped 1 .*Shaft
cmd loot spawn 1025 101 1025 loot create:blocks/cogwheel
wait 1
expect Dropped 1 .*Cogwheel
reject Unknown recipe|not found|ERROR|panicked
stop
