# P2a: Cindervane as a mod mob carried by a vanilla ravager, its state saved with it, running its
# own goals instead of the ravager's, fighting back and dying with its own loot.
# Run scripts/scenario.ps1 -Scenario tests/scenarios/p2a-dragons.txt -RunDir run/p2a-dragons -Fresh
mods target\mods\saints-dragons-roosts\saints-dragons.mmc
start
expect registered mob saintsdragons:cindervane \(carrier minecraft:ravager\)
botstart tester
wait 3
expect \[bot\] joined as tester
cmd op tester
cmd gamemode creative tester
cmd gamerule minecraft:spawn_mobs false
# Prevent unrelated saplings from decaying leaves contaminating the death-loot probe.
cmd gamerule minecraft:random_tick_speed 0
cmd tp tester 1025.5 101 1025.5
cmd forceload add 1015 1015 1035 1035
wait 12
cmd fill 1015 99 1015 1035 99 1035 minecraft:stone
await 30 Successfully filled|No blocks were filled
cmd fill 1015 100 1015 1035 110 1035 minecraft:air
await 30 Successfully filled|No blocks were filled
# Enclose the raised ground-combat arena: surrounding terrain can be dozens of
# blocks lower, and autonomous flight must not take the dragon outside it on restart.
cmd fill 1015 100 1015 1015 110 1035 minecraft:stone
await 30 Successfully filled|No blocks were filled
cmd fill 1035 100 1015 1035 110 1035 minecraft:stone
await 30 Successfully filled|No blocks were filled
cmd fill 1016 100 1015 1034 110 1015 minecraft:stone
await 30 Successfully filled|No blocks were filled
cmd fill 1016 100 1035 1034 110 1035 minecraft:stone
await 30 Successfully filled|No blocks were filled
cmd fill 1016 106 1016 1034 106 1034 minecraft:stone
await 30 Successfully filled|No blocks were filled
cmd time set midnight
wait 1

cmd cindervane spawn 1025 100 1025
wait 1
expect cindervane: spawned \d+, state set true
cmd cindervane find 1025 100 1025
wait 1
expect cindervane: found 1: saintsdragons:cindervane ravager state=wild ai=on
cmd cindervane find 1025 100 1100
wait 1
expect cindervane: found 0

# The type and the mod's state survive a restart.
bot quit
stop
start
botstart tester
wait 3
expect \[bot\] joined as tester
cmd tp tester 1025.5 101 1025.5
cmd time set midnight
wait 4
cmd cindervane find 1025 100 1025
wait 1
expect cindervane: found 1: saintsdragons:cindervane ravager state=wild ai=on

# Fall and lava don't hurt it, and nothing has hurt it yet, so it has no target.
cmd damage @e[type=minecraft:ravager,nbt={data:{moddedmc:{mob:"saintsdragons:cindervane"}}},limit=1] 500 minecraft:fall
cmd damage @e[type=minecraft:ravager,nbt={data:{moddedmc:{mob:"saintsdragons:cindervane"}}},limit=1] 500 minecraft:lava
wait 1
cmd cindervane find 1025 100 1025
wait 1
expect cindervane: found 1: .* target=none
# Hurt by a villager, it targets the villager, chases it and bites it to death (12 a bite).
cmd summon minecraft:villager 1031 100 1025
wait 1
cmd damage @e[type=minecraft:ravager,nbt={data:{moddedmc:{mob:"saintsdragons:cindervane"}}},limit=1] 1 minecraft:mob_attack by @e[type=minecraft:villager,limit=1]
wait 1
cmd cindervane find 1025 100 1025
wait 1
expect cindervane: found 1: .* target=villager
wait 10
# The magma volley may kill the villager far from the original arena. Verify its
# death, then return the surviving dragon to the query radius before checking its target.
await 5 \[bot\] sound: entity.villager.death
# The death sound precedes removal of the dying entity from the world.
wait 2
cmd tp @e[type=minecraft:ravager,nbt={data:{moddedmc:{mob:"saintsdragons:cindervane"}}},limit=1] 1025.5 100 1025.5
await 5 Teleported Ravager
cmd cindervane find 1025 100 1025
wait 1
expect cindervane: found 1: .* target=none; around: player$
# It dies with its own empty loot table, not the ravager's saddle.
cmd kill @e[type=minecraft:item,x=1009,y=84,z=1009,dx=32,dy=32,dz=32]
cmd damage @e[type=minecraft:ravager,nbt={data:{moddedmc:{mob:"saintsdragons:cindervane"}}},limit=1] 1000 minecraft:generic
await 5 Applied 1000 damage to Ravager
wait 3
cmd cindervane find 1025 100 1025
wait 1
expect cindervane: found 0: ; around: player$
reject (?i)panic|panicked
bot quit
stop
