# Raevyx breeding from 0.9.88: crouch-feeding a tame adult puts it in love. A tame female and male
# in love court for a few seconds and lay a Raevyx egg, and then can't breed again for a while.
mods target\mods\saints-dragons-roosts\saints-dragons.mmc
start
botstart tester
wait 3
cmd op tester
cmd gamemode creative tester
cmd gamerule minecraft:spawn_mobs false
cmd tp tester 1025.5 100 1029.5
cmd forceload add 1025 1025
wait 3
cmd fill 1013 99 1013 1037 99 1037 minecraft:stone
cmd fill 1013 100 1013 1037 110 1037 minecraft:air
cmd saintsdragons spawn raevyx 1022 100 1025
cmd saintsdragons spawn raevyx 1028 100 1025
wait 2.5
cmd data merge entity @e[type=minecraft:ravager,x=1022.5,y=100,z=1025.5,distance=..2,limit=1] {data:{"saintsdragons:appearance":{female:1b,variant:0b}}}
cmd data merge entity @e[type=minecraft:ravager,x=1028.5,y=100,z=1025.5,distance=..2,limit=1] {data:{"saintsdragons:appearance":{female:0b,variant:0b}}}
cmd item replace entity tester weapon.offhand with saintsdragons:creative_dragon_tame_meal
wait 0.5
bot use-entity ravager 1022.5 100 1025.5 off
wait 0.5
expect Dragon tamed
bot use-entity ravager 1028.5 100 1025.5 off
wait 0.5
expect Dragon tamed
# Food without crouching doesn't start breeding.
cmd item replace entity tester weapon.offhand with minecraft:mutton
bot input sneak
wait 0.5
bot use-entity ravager 1022.5 100 1025.5 off
wait 0.5
expect Raevyx is ready to breed!
bot use-entity ravager 1022.5 100 1025.5 off
wait 0.5
expect Raevyx is still eating
bot use-entity ravager 1028.5 100 1025.5 off
wait 0.5
expect Raevyx is ready to breed!
bot input none
wait 12
cmd fill 1013 95 1013 1037 110 1037 saintsdragons:raevyx_egg[hatch=1] replace saintsdragons:raevyx_egg[hatch=0]
wait 0.5
expect (?i)filled 1 block
cmd item replace entity tester weapon.offhand with minecraft:porkchop
bot input sneak
wait 0.5
cmd tp tester 1025.5 100 1029.5
wait 0.5
cmd execute at tester run tp @e[type=minecraft:ravager,limit=1,sort=nearest] 1025.5 100 1025.5
wait 0.5
bot use-entity ravager 1025.5 100 1025.5 off
wait 0.5
expect Raevyx is not ready to breed yet
bot input none
reject (?i)panic|panicked|\[bot\] error
bot quit
stop
