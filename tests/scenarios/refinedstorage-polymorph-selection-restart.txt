# A Pattern Grid chooses the emerald recipe among three live matches. The
# selected recipe identity is embedded in the Pattern, survives a clean server
# restart and drives the real Autocrafter result after redstone activation.
mods target\mods\refinedstorage.mmc
datapack tests\fixtures\polymorph-crafter-conflicts conflicts
start
expect moddedmc: loaded refinedstorage
cmd datapack enable "file/conflicts" last
wait 1
expect Loaded datapack 'conflicts'
stop
start
botstart tester
wait 3
cmd gamemode survival tester
cmd tp tester 1025.5 101 1023.5
wait 2
cmd fill 1021 100 1021 1033 105 1031 minecraft:air
cmd fill 1021 99 1021 1033 99 1031 minecraft:stone
cmd setblock 1025 100 1025 refinedstorage:controller
cmd setblock 1026 100 1025 refinedstorage:cable
cmd setblock 1027 100 1025 refinedstorage:disk_drive
cmd setblock 1025 100 1024 refinedstorage:grid
cmd setblock 1026 100 1026 refinedstorage:pattern_grid
cmd setblock 1025 100 1026 refinedstorage:autocrafter
cmd item replace entity tester hotbar.0 with refinedstorage:1k_storage_disk 1
bot hotbar 0
bot use 1027 100 1025
wait 0.5
expect Storage disk installed
cmd item replace entity tester hotbar.0 with minecraft:stone 1
bot use 1025 100 1024
wait 0.5
expect Stored 1 items
# Configure the Pattern Grid with one ghost stone, then select emerald.
cmd item replace entity tester hotbar.0 with minecraft:stone 1
bot input sneak
wait 0.5
bot use 1026 100 1026
wait 0.5
expect Crafting ghost ingredient recorded
bot input none
cmd item replace entity tester hotbar.0 with minecraft:air

wait 0.5

bot inventory-click 47 left
wait 0.5
bot expect slot 49 minecraft:diamond 1

wait 0.5

wait 0.5
expect Pattern encoded with actual recipe identity and output
bot use 1025 100 1026
wait 0.5
expect Autocrafter Pattern installed

wait 0.5

wait 0.5
expect Autocrafter recipe queued

stop
