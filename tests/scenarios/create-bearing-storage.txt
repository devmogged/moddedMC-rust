# Moving a chest and furnace keeps their saved inventories through a server restart.
mods target\mods\create.mmc
start
expect registered content block create:mechanical_bearing
botstart tester
wait 3
expect \[bot\] joined as tester
cmd tp tester 2289.5 101 2293.5 0 0
wait 2
cmd fill 2286 98 2286 2295 98 2295 minecraft:stone
cmd fill 2286 99 2286 2295 104 2295 minecraft:air
cmd setblock 2290 99 2290 create:creative_motor[facing=up]
wait 0.5
cmd kinetics 2290 99 2290 0
cmd setblock 2290 100 2290 create:mechanical_bearing[facing=up]
cmd setblock 2290 101 2290 minecraft:slime_block
cmd setblock 2291 101 2290 minecraft:chest[facing=east]
cmd setblock 2290 101 2289 minecraft:furnace[facing=north]
cmd item replace block 2291 101 2290 container.0 with minecraft:diamond 5
cmd item replace block 2290 101 2289 container.2 with minecraft:iron_ingot 3
cmd kinetics 2290 99 2290 1
wait 0.7
bot expect entity minecraft:block_display near 2290.5 101.5 2290.5 within 0.05 count 3
wait 0.5
expect ^\[bot\] expect entity ok
bot expect entity minecraft:item near 2290.5 101.5 2290.5 within 5 count 0
wait 0.5
expect ^\[bot\] expect entity ok
bot expect block 2291 101 2290 minecraft:air
wait 0.5
expect ^\[bot\] expect block ok 2291 101 2290
cmd save-all flush
stop
start
botstart tester
wait 2.5
expect \[bot\] joined as tester
bot expect entity minecraft:block_display near 2290.5 101.5 2290.5 within 0.05 count 3
wait 0.5
expect ^\[bot\] expect entity ok

# Turn the remaining quarter turn. The two containers must restore at their rotated positions.
cmd kinetics 2290 99 2290 15
wait 0.8
cmd kinetics 2290 99 2290 0
wait 0.5
bot expect block 2290 101 2289 minecraft:chest[facing=north]
wait 0.5
expect ^\[bot\] expect block ok 2290 101 2289
bot expect block 2289 101 2290 minecraft:furnace[facing=west]
wait 0.5
expect ^\[bot\] expect block ok 2289 101 2290
cmd data get block 2290 101 2289 Items[{Slot:0b}]
wait 0.5
expect following block data:.*minecraft:diamond
cmd data get block 2290 101 2289 Items[{Slot:0b}].count
wait 0.5
expect following block data: 5$
cmd data get block 2289 101 2290 Items[{Slot:2b}]
wait 0.5
expect following block data:.*minecraft:iron_ingot
cmd data get block 2289 101 2290 Items[{Slot:2b}].count
wait 0.5
expect following block data: 3$
bot expect entity minecraft:item near 2290.5 101.5 2290.5 within 5 count 0
wait 0.5
expect ^\[bot\] expect entity ok
reject (?i)panicked|trapped|\[bot\] error
stop
