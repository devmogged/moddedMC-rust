# WS-6 M1: all five source-2.2.4 fireResistant materials survive lava.
# Negative control: ordinary Chorus Crystal burns in lava, yet survives over dry stone.
# Each drop uses an isolated air-cleared arena and verifies empty inventory afterward.
mods target\mods\legendary-monsters.mmc
start
botstart fire_tester
wait 3
expect \[bot\] joined as fire_tester
# dinosaur_bone: fire-resistant lava
cmd tp fire_tester 10.5 101 10.5
wait 2
cmd fill 3 96 3 18 110 20 minecraft:air
cmd fill 3 95 3 18 95 20 minecraft:stone
cmd fill 3 96 3 18 96 20 minecraft:lava
cmd setblock 10 100 10 minecraft:stone
bot look 0 0
wait 1
cmd give fire_tester legendary_monsters:dinosaur_bone 1
wait 1
expect Gave 1
bot inventory-click 36 drop
wait 0.4
bot expect slot 36 minecraft:air 0
wait 0.2
expect \[bot\] expect inventory ok slot 36
wait 12
bot expect entity minecraft:item near 10.5 97 13 within 11 count 1
wait 1
expect \[bot\] expect entity ok minecraft:item
