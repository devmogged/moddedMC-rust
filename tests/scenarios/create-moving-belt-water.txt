mods target\mods\create.mmc
start
botstart tester
wait 3
expect \[bot\] joined as tester
cmd tp tester 2642.5 101 2642.5
wait 2
cmd fill 2636 97 2636 2647 97 2647 minecraft:stone
cmd fill 2636 98 2636 2647 107 2647 minecraft:air
cmd setblock 2640 98 2640 create:creative_motor[facing=up]
wait 0.5
cmd kinetics 2640 98 2640 0
cmd setblock 2640 99 2640 create:mechanical_bearing[facing=up]
cmd setblock 2640 100 2640 minecraft:slime_block
cmd setblock 2640 101 2640 create:shaft[axis=x,waterlogged=false]
cmd setblock 2640 101 2641 create:shaft[axis=x,waterlogged=true]
cmd setblock 2640 101 2642 create:shaft[axis=x,waterlogged=false]
cmd item replace entity tester hotbar.0 with create:belt_connector
wait 0.5
bot use 2640 101 2640
wait 0.5
bot use 2640 101 2642
wait 0.5
cmd kinetics 2640 101 2641
wait 0.5
expect create:belt\[facing=south,part=pulley,slope=horizontal,waterlogged=true\] at 2640 101 2641

cmd kinetics 2640 98 2640 15
wait 0.45
bot expect block 2640 101 2641 minecraft:water[level=0]
cmd setblock 2642 101 2640 minecraft:water
wait 0.5
cmd kinetics 2640 98 2640 0
wait 0.5
expect ^\[bot\] expect block ok 2640 101 2641 water\[level=0\]
cmd kinetics 2641 101 2640
cmd kinetics 2642 101 2640
wait 0.5
expect create:belt\[facing=east,part=pulley,slope=horizontal,waterlogged=false\] at 2641 101 2640
expect create:belt\[facing=east,part=end,slope=horizontal,waterlogged=true\] at 2642 101 2640
reject (?i)panicked|trapped|\[bot\] error
