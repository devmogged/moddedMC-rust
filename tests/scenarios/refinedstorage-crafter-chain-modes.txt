# its four speed upgrades, pulse mode and powered-edge checkpoint persist.
mods target\mods\refinedstorage.mmc
start
expect moddedmc: loaded refinedstorage
botstart tester
wait 3
cmd gamemode survival tester
cmd tp tester 1025.5 101 1023.5
wait 2
cmd fill 1021 100 1021 1033 105 1031 minecraft:air
cmd fill 1021 99 1021 1033 99 1031 minecraft:stone

cmd setblock 1026 100 1025 refinedstorage:cable
cmd setblock 1027 100 1025 refinedstorage:crafter[facing=east]
cmd setblock 1028 100 1025 refinedstorage:crafter[facing=east]
wait 1
# Install four Speed upgrades on the root Crafter.
cmd item replace entity tester hotbar.0 with refinedstorage:speed_upgrade 4
bot hotbar 0

wait 0.3
bot use 1028 100 1025
wait 0.3
bot use 1028 100 1025
wait 0.3
bot use 1028 100 1025
wait 0.5
expect Crafter speed upgraded

wait 0.5
