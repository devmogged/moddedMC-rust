# Ivy's open_trade dialogue nodes open her trades. The first time, she rolls the released
# ivy_trades table: the fixed Hearty Dragon Meal trade and four of the six dragon egg trades. Her
# villager carrier then keeps and saves them.
mods target\mods\saints-dragons-roosts\saints-dragons.mmc
start
wait 1
botstart tester
wait 3
cmd gamemode creative tester
cmd gamerule minecraft:spawn_mobs false
cmd tp tester 1026.5 224 1027.5 0 0
cmd forceload add 1025 1025
wait 4
cmd fill 1024 223 1024 1038 223 1038 minecraft:stone
cmd fill 1024 224 1024 1038 232 1038 minecraft:air
wait 3
cmd item replace entity tester hotbar.0 with saintsdragons:ivy_the_merchant_spawn_egg
wait 0.5
bot use 1027 223 1026 up
wait 0.5
cmd execute at tester run data merge entity @e[type=minecraft:villager,limit=1,sort=nearest] {NoAI:1b,Tags:["ivy_probe"]}
cmd item replace entity tester hotbar.0 with minecraft:air
wait 0.5
bot use-entity villager 1027.5 224 1026.5 main
wait 0.5
expect \[bot\] dialog: .*Ivy the Merchant
# first_meeting/intro -> first_meeting/c_trader
bot click saintsdragons:ivy/1/2
wait 0.5
expect \[bot\] dialog: .*Ivy the Merchant
# first_meeting/c_trader -> first_meeting/c_trade_straight, an open_trade node
bot click saintsdragons:ivy/2/1
wait 0.5
expect \[bot\] dialog: .*Ivy the Merchant
bot click saintsdragons:ivy/3/0
wait 1
# Pumpkin's data paths don't reach Offers, so read the whole entity.
cmd data get entity @e[tag=ivy_probe,limit=1]
wait 0.5
expect Offers: \{Recipes: \[\{buy: \{components: \{\}, count: 4, id: "minecraft:egg"\}, buyB: \{components: \{\}, count: 4, id: "minecraft:salmon"\}, demand: 0, maxUses: 9999, priceMultiplier: 0f, rewardExp: 0b, sell: \{components: \{\}, count: 6, id: "saintsdragons:hearty_dragon_meal"\}
cmd data get entity @e[tag=ivy_probe,limit=1]
wait 0.5
expect Offers: \{Recipes: \[\{buy: ([^\]]*?\}, \{buy: ){4}[^\]]*\}\]\}
bot quit
stop
