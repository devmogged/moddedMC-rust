# slot. Its contents appear in the Grid, persist restart, and are rescued as a
# filled tier-matched cartridge when the block is mined in survival.

start

cmd gamemode survival tester

cmd item replace entity tester hotbar.0 with minecraft:emerald 11

wait 0.5

cmd data get block 1027 100 1025
wait 0.5
