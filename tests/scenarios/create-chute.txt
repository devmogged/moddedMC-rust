# Create 6.0.10 Chute. Verify item handoffs from the stock client's view.
mods target\mods\create.mmc
start
expect registered content block create:chute
botstart tester
wait 3
expect \[bot\] joined as tester
cmd tp tester 2082.5 101 2012.5 0 0
wait 2
cmd fill 2078 99 2008 2085 99 2019 minecraft:stone
cmd fill 2078 100 2008 2085 105 2019 minecraft:air
cmd tp tester 2082.5 100 2012.5 0 0
cmd gamemode survival tester

# A falling item goes through two chutes and appears on the depot.
cmd setblock 2081 100 2012 create:depot
cmd setblock 2081 101 2012 create:chute
cmd setblock 2081 102 2012 create:chute
wait 0.5
bot expect block 2081 102 2012 minecraft:note_block[powered=true]
wait 0.5
expect ^\[bot\] expect block ok
cmd summon minecraft:item 2081.5 104 2012.5 {Item:{id:"minecraft:copper_ingot",count:3},PickupDelay:32767s}
wait 1.5
bot expect entity minecraft:item_display near 2081.5 100.9 2012.5 within 0.24 count 1
wait 0.5
expect ^\[bot\] expect entity ok
bot use 2081 100 2012
wait 0.5
bot expect slot 36 minecraft:copper_ingot 3
wait 0.5
expect ^\[bot\] expect inventory ok
