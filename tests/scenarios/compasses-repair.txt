# Zero-cost config, actual shapeless repair in the stock player 2x2 grid.
# Nature's recipe returns a new blank compass, not the original located stack.
mods target\mods\compasses.mmc
start
cmd difficulty peaceful
botstart tester
wait 3
expect \[bot\] joined as tester
cmd gamemode survival tester
cmd experience set tester 7 levels
cmd experience set tester 3 points
cmd item replace entity tester hotbar.0 with naturescompass:naturescompass
cmd item replace entity tester hotbar.1 with minecraft:compass
bot hotbar 0
wait 1
bot use-item main
wait 1
expect \[bot\] dialog: .*Nature's Compass
bot click-label minecraft:birch_forest
wait 2
expect Compass search started
await 15 Found minecraft:birch_forest at X -287, Z 6 in minecraft:overworld
bot expect tracker 36 minecraft:overworld -287 6
wait 1
expect \[bot\] expect inventory ok tracker 36 minecraft:overworld -287 6
cmd experience query tester levels
cmd experience query tester points
wait 1
expect-all tester has 7 experience levels && tester has 3 experience points
bot use-item main
wait 1
expect \[bot\] dialog: .*Nature's Compass
bot inventory-click 36 left
wait 0.5
bot inventory-click 1 left
wait 0.5
bot inventory-click 37 left
wait 0.5
bot inventory-click 2 left
wait 1
bot expect slot 0 minecraft:compass 1
wait 1
reject \[bot\] error:|panicked
expect \[bot\] expect inventory ok slot 0 compass 1
bot inventory-click 0 left
wait 0.5
bot inventory-click 36 left
wait 0.5
bot expect slot 0 minecraft:air 0
bot expect slot 1 minecraft:air 0
bot expect slot 2 minecraft:air 0
bot expect slot 36 minecraft:compass 1
wait 1
reject \[bot\] error:|panicked
expect-all \[bot\] expect inventory ok slot 0 air 0 && \[bot\] expect inventory ok slot 1 air 0 && \[bot\] expect inventory ok slot 2 air 0 && \[bot\] expect inventory ok slot 36 compass 1
# Player inventory uses window zero; there is no separate container to close.
bot expect tracker 36 absent
bot expect slot 37 minecraft:air 0
bot expect cursor minecraft:air 0
wait 1
expect-all \[bot\] expect inventory ok tracker 36 absent && \[bot\] expect inventory ok slot 37 air 0 && \[bot\] expect inventory ok cursor air 0
cmd clear tester naturescompass:naturescompass 0
await 1 Found 1 matching item
bot click-label Find next / retry
wait 1
reject Compass search started
expect Compass or dimension changed
bot use-item main
wait 1
reject \[bot\] dialog button: Find next / retry ->
expect \[bot\] dialog: .*Select a target in this dimension
cmd experience query tester levels
cmd experience query tester points
wait 1
expect-all tester has 7 experience levels && tester has 3 experience points
bot quit
stop
