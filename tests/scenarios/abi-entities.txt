# moddedmc:entities/1 through the wisp mod: spawn a creature, move it, find it again after a
# restart (the world saves it, the mod saves its id).
# Run: scripts\scenario.ps1 -Scenario tests\scenarios\abi-entities.txt -RunDir run\abi-entities -Fresh
mods target\mods\wisp.mmc
start
wait 1
expect registered command /wisp
botstart tester
wait 3
expect \[bot\] joined as tester
cmd op tester
cmd tp tester 4097.5 100 4097.5
wait 3
cmd fill 4090 99 4090 4105 99 4105 minecraft:stone
cmd fill 4090 100 4090 4105 106 4105 minecraft:air
cmd tp tester 4097.5 100 4097.5
wait 2
bot cmd wisp
wait 2
expect \[bot\] chat: .*wisp: spawned at 4097.5 100.0 4097.5
bot cmd wispstep
wait 2
expect \[bot\] chat: .*wispstep: moved 1
bot cmd wisplist
wait 1
expect \[bot\] chat: .*wisplist: 4098.5 100.0 4097.5
# Entity events: the mod answers a right-click and cancels an attack.
bot interact
wait 1
expect \[bot\] chat: .*wisp: the wisp hums
bot attack
wait 1
expect \[bot\] chat: .*wisp: your hand passes through
bot attack
wait 1
bot cmd wisplist
wait 1
expect \[bot\] chat: .*wisplist: 4098.5 100.0 4097.5
reject (?i)panic|panicked
bot quit
stop

start
wait 1
botstart tester
wait 3
cmd tp tester 4097.5 100 4097.5
wait 3
bot cmd wisplist
wait 1
expect \[bot\] chat: .*wisplist: 4098.5 100.0 4097.5
# AI hooks: a chicken with its AI off stays put until the mod moves it.
cmd tp tester 4094.5 100 4094.5
wait 2
bot cmd wisppet
wait 1
expect \[bot\] chat: .*wisppet: ai off true
wait 8
bot cmd wisplist
wait 1
expect \[bot\] chat: .*wisplist: 4098.5 100.0 4097.5, 4094.5 100.0 4094.5
bot cmd wispstep
wait 1
expect \[bot\] chat: .*wispstep: moved 2
bot cmd wisplist
wait 1
expect \[bot\] chat: .*wisplist: 4099.5 100.0 4097.5, 4095.5 100.0 4094.5
# Pathfinding: the pet walks to a target the mod picks.
bot cmd wispgo
wait 1
expect \[bot\] chat: .*wispgo: walking 1
wait 8
bot cmd wisparrived
wait 1
expect \[bot\] chat: .*wisparrived: 1 of 1
reject (?i)panic|panicked
bot quit
stop
